from ontology.types import Equipment, YieldResult def calc_equipment_yield(equipment: Equipment, window_days: int = 30) -> YieldResult: """统计设备近 N 天的产出与不良，计算综合良率""" reports = equipment.quality_reports.where( created_at >= now() - timedelta(days=window_days)) total = reports.count() good = reports.where(grade == QualityGrade.A).count() \ + reports.where(grade == QualityGrade.B).count() return YieldResult( yield_rate=round(good / total * 100, 2) if total else None, sample_size=total, equipment_id=equipment.equipment_id, )
